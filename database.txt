CREATE TABLE public.profiles (
  id uuid NOT NULL,
  username text,
  phone_number text,
  avatar_url text,
  role text CHECK (role = ANY (ARRAY['pemilik_kucing'::text, 'admin'::text, 'super_admin'::text])),
  is_active boolean DEFAULT true,
  created_by uuid,
  subscription_tier text DEFAULT 'free'::text CHECK (subscription_tier = ANY (ARRAY['free'::text, 'trial'::text, 'premium'::text])),
  subscription_expires_at timestamp with time zone,
  created_at timestamp with time zone DEFAULT now(),
  is_blocked boolean NOT NULL DEFAULT false,
  onboarding_completed boolean DEFAULT false,
  full_name text,
  CONSTRAINT profiles_pkey PRIMARY KEY (id),
  CONSTRAINT profiles_id_fkey FOREIGN KEY (id) REFERENCES auth.users(id),
  CONSTRAINT profiles_created_by_fkey FOREIGN KEY (created_by) REFERENCES public.profiles(id)
);
CREATE TABLE public.cats (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  owner_id uuid NOT NULL,
  name text NOT NULL,
  photo_url text,

  birth_date date,
  gender text CHECK (gender = ANY (ARRAY['jantan'::text, 'betina'::text])),
  is_neutered boolean DEFAULT false,
  created_at timestamp with time zone DEFAULT now(),
  weight numeric,
  CONSTRAINT cats_pkey PRIMARY KEY (id),
  CONSTRAINT cats_owner_id_fkey FOREIGN KEY (owner_id) REFERENCES public.profiles(id)
);
